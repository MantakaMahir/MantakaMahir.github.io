flowchart LR
    start([Start]) --> paired{Pendant already paired?}
    paired -- Yes --> home["Home — Live Posture"]
    paired -- No --> welcome["Welcome"]

    welcome --> permissions["Permission Explanation"]
    permissions --> granted{Bluetooth permission granted?}
    granted -- Yes --> pairing["Find and Pair"]
    granted -- No --> permhelp["Permission Help / Open Settings"]
    permhelp --> permissions

    pairing --> found{Pendant found?}
    found -- Yes --> connect["Connect Pendant"]
    found -- No --> searching["Searching / No Device"]
    searching --> scanagain["Scan Again"]
    scanagain --> pairing
    searching --> troubleshoot["Troubleshooting"]
    troubleshoot --> pairing

    connect --> connected{Connected?}
    connected -- Yes --> wear["Wear Guidance"]
    connected -- No --> connretry["Connection Retry"]
    connretry --> pairing

    wear --> positioned{Pendant positioned correctly?}
    positioned -- Yes --> calibration["Calibration — parameters per firmware"]
    positioned -- No --> tips["Placement Tips"]
    tips --> wear

    calibration --> calibrated{Calibration complete?}
    calibrated -- Yes --> home
    calibrated -- No --> recovery["Calibration Recovery / Retry"]
    recovery --> calibration

    home --> posture{Correct posture detected?}
    posture -- Yes --> upright["App shows upright posture state"]
    posture -- No --> adjust["App prompts user to adjust posture"]
    upright --> haptic{Haptic alert confirmed for MVP?}
    adjust --> haptic
    haptic -- Yes --> hapticAlert["Pendant haptic alert per firmware"]
    haptic -- No --> appFeedback["App visual feedback only"]
    hapticAlert --> appFeedback
    appFeedback --> home

    home --> loss{Pendant connection lost during use?}
    loss -- Yes --> lost["Connection Lost — show status in app"]
    lost --> reconnect["Attempt reconnection"]
    reconnect --> reconnected{Connection restored?}
    reconnected -- Yes --> home
    reconnected -- No --> recoverConnection["Connection Recovery / Retry"]
    recoverConnection --> reconnect
    loss -- No --> home

    home <-->|Bottom navigation| history["History"]
    home <-->|Bottom navigation| device["Device"]
    home <-->|Bottom navigation| settings["Settings"]
    home --> wellness["Wellness Readings — Heart Rate / SpO2 only if confirmed for MVP"]
    wellness -->|Back to Home| home

    device --> status["Connection, Battery and Sync Status"]
    status --> lost
    device --> recalibrate["Recalibrate"]
    recalibrate --> calibration
    device --> replace{Replace pendant?}
    replace -- Yes --> pairing
    replace -- No --> device

    settings --> reminders["Notifications and Quiet Hours"]
    settings --> preferences["Language and Appearance"]
    settings --> delete{Delete local data?}
    delete -- No --> settings
    delete -- Yes --> confirm["Confirm Delete"]
    confirm --> cleared([Data Cleared])
